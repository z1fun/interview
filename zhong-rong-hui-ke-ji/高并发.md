## 一、Go 并发核心机制

### 1.1 GMP 调度模型（面试高频！）

```
┌─────────────────────────────────────────────┐
│  G (Goroutine)   — 用户态轻量线程，2KB起    │
│  M (Machine)     — 系统线程，执行G的载体     │
│  P (Processor)   — 逻辑处理器，G的调度上下文  │
│                    默认=CPU核数(GOMAXPROCS)  │
└─────────────────────────────────────────────┘

调度关系：
  每个P有本地G队列（256容量）
  M必须绑定P才能执行G
  G阻塞时（系统调用/网络IO），M与P分离，P找新M继续工作
  
关键机制：
  - work stealing：P的本地队列空了，从其他P偷一半G
  - hand off：G阻塞时，M带走G，P接管新的G继续运行
  - 抢占式调度：基于信号的抢占（Go 1.14+），防止一个G霸占CPU
```

**面试必答 GMP vs OS线程**：
> "Go的并发优势在于GMP模型：(1) Goroutine是用户态调度，创建成本极低（~2KB），而系统线程1MB起；(2) GOMAXPROCS数量的P充分利用多核；(3) work stealing自动负载均衡；(4) 网络IO异步，不会独占线程；(5) 简单用 `go` 关键字即可启动，心智负担低。"

### 1.2 Goroutine 使用模式

```go
// 1. 基础并发
go doWork()

// 2. sync.WaitGroup 等待多个goroutine完成
var wg sync.WaitGroup
for i := 0; i < 10; i++ {
    wg.Add(1)
    go func(id int) {
        defer wg.Done()
        process(id)
    }(i) // ⚠️ 注意：传参避免闭包捕获循环变量
}
wg.Wait()

// 3. errgroup：并发执行+错误处理（关键！）
import "golang.org/x/sync/errgroup"

g, ctx := errgroup.WithContext(ctx)
g.SetLimit(10) // 限制并发数
for _, item := range items {
    item := item
    g.Go(func() error {
        return process(ctx, item)
    })
}
if err := g.Wait(); err != nil {
    // 处理第一个错误
}

// 4. 生产者-消费者模式
func producerConsumer() {
    jobs := make(chan Job, 100)   // 带缓冲
    results := make(chan Result, 100)

    // 启动worker池
    for w := 0; w < 10; w++ {
        go worker(w, jobs, results)
    }

    // 生产者
    go func() {
        for _, job := range allJobs {
            jobs <- job
        }
        close(jobs)
    }()

    // 消费者
    for result := range results {
        handle(result)
    }
}
```

### 1.3 Channel 通信

```go
// "不要通过共享内存来通信，而要通过通信来共享内存" — Go Proverbs

// 无缓冲 channel：同步
ch := make(chan int)
go func() { ch <- 42 }()
v := <-ch // 阻塞直到收到数据

// 有缓冲 channel：异步，缓冲满之前不阻塞
ch := make(chan int, 100)

// select 多路复用（控制多个goroutine生命周期）
select {
case msg := <-msgCh:
    handle(msg)
case <-ctx.Done():
    return ctx.Err() // 超时/取消
case <-ticker.C:
    doPeriodic()     // 定时任务
}

// channel 正确关闭原则（面试常常问！）
// ✅ 发送方关闭channel，接收方不关闭
// ✅ 不要向已关闭的channel发送数据 → panic
// ✅ 从已关闭的channel读取 → 返回零值, false
// ✅ nil channel 读写永久阻塞，select中永远不命中
```

### 1.4 sync 包关键组件

```go
// Mutex — 互斥锁
var mu sync.Mutex
mu.Lock()
// critical section
mu.Unlock()

// RWMutex — 读写锁（读多写少场景比Mutex性能好）
var rw sync.RWMutex
rw.RLock()  // 多个goroutine可同时持有读锁
rw.RUnlock()
rw.Lock()   // 写锁独占
rw.Unlock()

// Once — 单次执行（单例初始化首选）
var once sync.Once
once.Do(func() {
    // 只执行一次，所有goroutine都等到它执行完
    initResource()
})

// Pool — 对象池（复用高频创建的对象）
var bufPool = sync.Pool{
    New: func() interface{} { return new(bytes.Buffer) },
}
buf := bufPool.Get().(*bytes.Buffer)
buf.Reset()
defer bufPool.Put(buf)

// sync.Map — 并发安全Map（适合读多写少/key变化少的场景）
var sm sync.Map
sm.Store("key", "value")
v, ok := sm.Load("key")
```

## 二、限流策略

### 2.1 令牌桶（Token Bucket）— Go官方实现

```go
import "golang.org/x/time/rate"

// 每秒产生100个令牌，桶容量200（允许瞬间突发）
limiter := rate.NewLimiter(100, 200)

func handler(w http.ResponseWriter, r *http.Request) {
    if !limiter.Allow() {
        http.Error(w, "Too Many Requests", 429)
        return
    }
    // 正常处理
}

// Wait方式：阻塞直到获取令牌
func handlerWait(w http.ResponseWriter, r *http.Request) {
    if err := limiter.Wait(r.Context()); err != nil {
        http.Error(w, "Rate limited", 429)
        return
    }
    // 正常处理
}
```

### 2.2 漏桶（Leaky Bucket）— 平滑流量

```go
// 漏桶 = 固定速率处理，平滑突刺流量
// Go中使用 rate.Limiter(0) burst 或使用 time.Ticker 实现

func leakyBucket(requests <-chan Request, rate int) <-chan Response {
    ticker := time.NewTicker(time.Second / time.Duration(rate))
    defer ticker.Stop()

    for req := range requests {
        <-ticker.C // 严格控制处理速度
        go process(req)
    }
}
```

### 2.3 滑动窗口 — Redis实现

```go
// 使用Redis有序集合实现滑动窗口计数
const script = `
local key = KEYS[1]
local window = tonumber(ARGV[1])          -- 窗口大小(秒)
local limit = tonumber(ARGV[2])           -- 限制次数
local now = tonumber(ARGV[3])             -- 当前时间戳

-- 移除窗口外的记录
redis.call('ZREMRANGEBYSCORE', key, 0, now - window * 1000)

-- 当前窗口内请求数
local current = redis.call('ZCARD', key)

if current < limit then
    redis.call('ZADD', key, now, now .. '-' .. math.random())
    redis.call('PEXPIRE', key, windows * 1000)
    return 1  -- 放行
else
    return 0  -- 限流
end
`

// Go代码调用
func (s *RateLimiter) Allow(ctx context.Context, key string, window time.Duration, limit int) bool {
    now := time.Now().UnixMilli()
    result, _ := s.redis.Eval(ctx, script,
        []string{fmt.Sprintf("ratelimit:%s", key)},
        window.Seconds(), limit, now,
    ).Int()
    return result == 1
}
```

## 三、熔断与降级

### 3.1 熔断器模式（Circuit Breaker）

```
三种状态：
  CLOSED(正常) → 失败次数达阈值 → OPEN(熔断，快速失败)
  OPEN → 超时窗口过 → HALF_OPEN(尝试放行少量请求)
  HALF_OPEN → 成功 → CLOSED | 失败 → OPEN
```

```go
// go-zero内置熔断器（你在POS数据贯通中应该已经用过了！）
// go-zero的breaker基于Google SRE的"自适应限流"算法

import "github.com/zeromicro/go-zero/core/breaker"

// 使用示例（go-zero中的实际用法）
brk := breaker.NewBreaker()
err := brk.DoWithAcceptable(func() error {
    return callDownstream()
}, func(err error) bool {
    // 只统计服务端错误，业务错误不计入熔断
    return err == ErrServiceUnavailable
})

// 面试话术：
// "我在go-zero项目中使用过熔断器，它的原理是基于滑动窗口统计错误率，
// 当错误率超过阈值时自动熔断(hystrix是固定阈值，go-zero是动态自适应的)。
// 熔断期间直接返回降级响应，保护下游服务不被雪崩效应打垮。"
```

### 3.2 降级策略

```go
// 降级 = 非核心功能在高峰时选择性关闭
type DegradeLevel int
const (
    DegradeNone  DegradeLevel = iota
    DegradeLight              // 关闭日志详情
    DegradeMedium             // 关闭推荐/广告
    DegradeHeavy              // 仅保留核心支付链路
)

func (s *Service) GetOrderDetail(ctx context.Context, orderID string) (*OrderDetail, error) {
    // 核心信息：订单基本信息（不可降级）
    order, err := s.db.QueryOrder(ctx, orderID)
    if err != nil {
        return nil, err
    }

    detail := &OrderDetail{Order: order}

    // 非核心信息：商品推荐（可降级）
    if s.degrade <= DegradeMedium {
        detail.Recommendations, _ = s.recommendService.Get(ctx, orderID)
    }

    return detail, nil
}
```

## 四、缓存策略（性能优化的核心）

### 4.1 缓存模式对比

| 模式 | 描述 | 适用场景 |
|------|------|----------|
| **Cache-Aside** | 业务代码控制，旁路缓存 | 最常用，读多写少 |
| **Read-Through** | 缓存负责读DB，业务只读缓存 | 缓存层封装 |
| **Write-Through** | 同步写缓存+DB | 数据强一致 |
| **Write-Behind** | 异步写DB，先写缓存 | 写密集型，有丢失风险 |

### 4.2 Cache-Aside 标准实现

```go
// 读：先读缓存 → miss → 读DB → 回写缓存
func (s *Service) GetOrder(ctx context.Context, orderID string) (*Order, error) {
    // 1. 查缓存
    key := fmt.Sprintf("order:%s", orderID)
    var order Order
    err := s.redis.Get(ctx, key).Scan(&order)
    if err == nil {
        return &order, nil
    }
    if err != redis.Nil {
        return nil, err
    }

    // 2. Cache miss，查DB（加单飞防止缓存击穿）
    loaded, err, _ := s.singleFlight.Do(key, func() (interface{}, error) {
        return s.db.QueryOrder(ctx, orderID)
    })
    if err != nil {
        // 3. 缓存空值防止穿透（短TTL）
        s.redis.Set(ctx, key, "NULL", 1*time.Minute)
        return nil, err
    }

    order = *loaded.(*Order)
    // 4. 回写缓存
    s.redis.Set(ctx, key, &order, 10*time.Minute)
    return &order, nil
}

// 写：先写DB → 删缓存（Cache-Aside经典做法）
func (s *Service) UpdateOrder(ctx context.Context, order *Order) error {
    // 1. 更新DB
    if err := s.db.UpdateOrder(ctx, order); err != nil {
        return err
    }
    // 2. 删除缓存（不更新，防并发写乱序问题）
    key := fmt.Sprintf("order:%s", order.ID)
    s.redis.Del(ctx, key)
    return nil
}
```

### 4.3 缓存三大问题与解法

```
┌──────────────┬──────────────────────────┬──────────────────────┐
│ 问题         │ 现象                      │ 解法                  │
├──────────────┼──────────────────────────┼──────────────────────┤
│ 缓存穿透     │ 查不存在的数据，          │ ① 布隆过滤器          │
│              │ 每次都穿透到DB            │ ② 缓存空值(短TTL)    │
├──────────────┼──────────────────────────┼──────────────────────┤
│ 缓存击穿     │ 热点key过期，             │ ① singleflight       │
│              │ 瞬时大量请求打到DB         │ ② 互斥锁             │
│              │                          │ ③ 永不过期+异步更新   │
├──────────────┼──────────────────────────┼──────────────────────┤
│ 缓存雪崩     │ 大量key同时过期，         │ ① TTL加随机值         │
│              │ DB瞬间被压垮              │ ② 多级缓存            │
│              │                          │ ③ 限流降级            │
└──────────────┴──────────────────────────┴──────────────────────┘
```

### 4.4 singleflight（防止缓存击穿）

```go
import "golang.org/x/sync/singleflight"

var g singleflight.Group

func (s *Service) GetItem(ctx context.Context, id string) (*Item, error) {
    key := fmt.Sprintf("item:%s", id)

    // 缓存查询
    if item, ok := s.cache.Get(key); ok {
        return item, nil
    }

    // 只有一个goroutine会真正执行fn，其他等待共享结果
    v, err, _ := g.Do(key, func() (interface{}, error) {
        return s.db.QueryItem(ctx, id)
    })
    if err != nil {
        return nil, err
    }

    item := v.(*Item)
    s.cache.Set(key, item, 10*time.Minute)
    return item, nil
}
```

> "高并发的核心挑战：(1) 有限资源处理大量请求——通过限流保障核心链路；(2) 服务间调用失败——通过熔断降级防止级联故障；(3) 数据库压力——通过多级缓存(本地缓存+Redis+DB)减少穿透；(4) 利用Go的GMP调度模型和goroutine低成本特性，用协程池模式控制并发数，避免资源耗尽。在之前的POS数据贯通项目中，我们通过go-zero微服务+etcd+消息队列实现了高并发数据处理。"
